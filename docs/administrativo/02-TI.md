## Tabelas SQL

```sql
-- CHAMADOS DE TI
CREATE TABLE it_tickets (
    id INT PRIMARY KEY AUTO_INCREMENT,
    requester_id INT NOT NULL,
    ticket_date DATETIME NOT NULL,
    category ENUM('hardware','software','network','email','system','telephony','access','other'),
    priority ENUM('low','medium','high','urgent'),
    subject VARCHAR(200) NOT NULL,
    description TEXT,
    assigned_to INT,
    solution TEXT,
    closed_date DATETIME,
    status ENUM('open','in_progress','waiting','resolved','closed'),
    created_at DATETIME,
    updated_at DATETIME
);

-- LICENÇAS DE SOFTWARE
CREATE TABLE software_licenses (
    id INT PRIMARY KEY AUTO_INCREMENT,
    software_name VARCHAR(200) NOT NULL,
    license_type ENUM('perpetual','subscription','free'),
    vendor VARCHAR(100),
    license_key VARCHAR(255),
    seats INT DEFAULT 1,
    purchase_date DATE,
    renewal_date DATE,
    cost DECIMAL(10,2),
    status ENUM('active','expired','canceled'),
    created_at DATETIME
);
